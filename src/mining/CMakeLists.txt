# GPU dependencies are private and optional; ordinary Core/Qt headers never
# import vendor SDKs. OpenCL is loaded at runtime; CUDA uses the static runtime.
option(WITH_CUDA_MINING "Build the optional NVIDIA CUDA MercaHash backend" OFF)
option(WITH_OPENCL_MINING "Build the optional AMD OpenCL MercaHash backend" OFF)
target_sources(bitcoin_node PRIVATE gpu_backend.cpp)
if(WITH_CUDA_MINING OR WITH_OPENCL_MINING)
  if(APPLE)
    message(FATAL_ERROR "Mercatura GPU mining currently supports Linux and Windows; use CPU-only on macOS")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "GPU mining requires a 64-bit host toolchain")
  endif()
endif()
if(WITH_CUDA_MINING)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(NOT "$ENV{CUDAARCHS}" STREQUAL "")
      set(CMAKE_CUDA_ARCHITECTURES "$ENV{CUDAARCHS}")
    else()
      set(CMAKE_CUDA_ARCHITECTURES 75-real 75-virtual)
    endif()
  endif()
  # Default to Turing and newer, including forward-compatible PTX. CUDA 12
  # builds can explicitly include 52/60/61 for older devices. The configured
  # minimum must be reflected in discovery rather than advertising unsupported
  # binaries to an older physical device.
  set(MERCATURA_CUDA_MIN_ARCH 999)
  set(MERCATURA_CUDA_MIN_PTX_ARCH 999)
  foreach(architecture IN LISTS CMAKE_CUDA_ARCHITECTURES)
    if(NOT "${architecture}" MATCHES "^[0-9]+(-real|-virtual)?$")
      message(FATAL_ERROR "Set explicit numeric CMAKE_CUDA_ARCHITECTURES, optionally suffixed -real/-virtual, for auditable GPU capability detection")
    endif()
    string(REGEX MATCH "^[0-9]+" numeric_arch "${architecture}")
    if(numeric_arch AND numeric_arch LESS MERCATURA_CUDA_MIN_ARCH)
      set(MERCATURA_CUDA_MIN_ARCH ${numeric_arch})
    endif()
    if(NOT "${architecture}" MATCHES "-real$" AND numeric_arch LESS MERCATURA_CUDA_MIN_PTX_ARCH)
      set(MERCATURA_CUDA_MIN_PTX_ARCH ${numeric_arch})
    endif()
  endforeach()
  if(MERCATURA_CUDA_MIN_ARCH EQUAL 999)
    message(FATAL_ERROR "Set explicit numeric CMAKE_CUDA_ARCHITECTURES for auditable GPU capability detection")
  endif()
  if(MERCATURA_CUDA_MIN_PTX_ARCH GREATER MERCATURA_CUDA_MIN_ARCH)
    message(FATAL_ERROR "NVIDIA discovery requires forward-compatible PTX at the lowest configured architecture ${MERCATURA_CUDA_MIN_ARCH}. Include '${MERCATURA_CUDA_MIN_ARCH}-virtual' or plain '${MERCATURA_CUDA_MIN_ARCH}' in CMAKE_CUDA_ARCHITECTURES; real-only binaries cannot support every newer GPU major architecture.")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 12.0 REQUIRED)
  add_library(mercatura_cuda STATIC cuda_backend.cu)
  set_target_properties(mercatura_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Static)
  target_include_directories(mercatura_cuda PRIVATE ${PROJECT_SOURCE_DIR}/src)
  target_compile_definitions(mercatura_cuda PRIVATE MERCATURA_CUDA MERCATURA_CUDA_MIN_ARCH=${MERCATURA_CUDA_MIN_ARCH})
  if(WIN32)
    target_compile_definitions(mercatura_cuda PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  endif()
  target_link_libraries(mercatura_cuda PRIVATE CUDA::cudart_static)
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/gpu_backend.cpp TARGET_DIRECTORY bitcoin_node
    APPEND PROPERTY COMPILE_DEFINITIONS MERCATURA_CUDA)
  target_link_libraries(bitcoin_node PRIVATE mercatura_cuda)
endif()
if(WITH_OPENCL_MINING)
  find_package(OpenCL 1.2 REQUIRED)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mercahash_gpu.h" MERCAHASH_GPU_SOURCE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS mercahash_gpu.h)
  # MSVC limits each individual string literal to 16,380 bytes. Adjacent raw
  # literals retain the exact device source while keeping each below that limit.
  string(PREPEND MERCAHASH_GPU_SOURCE "#define MH_OPENCL\n")
  string(FIND "${MERCAHASH_GPU_SOURCE}" ")MERCAHASH_CL\"" delimiter_position)
  if(NOT delimiter_position EQUAL -1)
    message(FATAL_ERROR "OpenCL device source contains its embedding delimiter")
  endif()
  string(LENGTH "${MERCAHASH_GPU_SOURCE}" source_length)
  set(source_offset 0)
  set(MERCAHASH_GPU_LITERALS "")
  while(source_offset LESS source_length)
    string(SUBSTRING "${MERCAHASH_GPU_SOURCE}" ${source_offset} 8192 source_chunk)
    string(APPEND MERCAHASH_GPU_LITERALS "R\"MERCAHASH_CL(${source_chunk})MERCAHASH_CL\"\n")
    math(EXPR source_offset "${source_offset} + 8192")
  endwhile()
  configure_file(opencl_source.h.in ${PROJECT_BINARY_DIR}/src/mining/opencl_source.h @ONLY)
  target_sources(bitcoin_node PRIVATE opencl_backend.cpp)
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/opencl_backend.cpp TARGET_DIRECTORY bitcoin_node
    APPEND PROPERTY INCLUDE_DIRECTORIES ${OpenCL_INCLUDE_DIRS})
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/opencl_backend.cpp TARGET_DIRECTORY bitcoin_node
    APPEND PROPERTY COMPILE_DEFINITIONS MERCATURA_OPENCL)
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/gpu_backend.cpp TARGET_DIRECTORY bitcoin_node
    APPEND PROPERTY COMPILE_DEFINITIONS MERCATURA_OPENCL)
  target_link_libraries(bitcoin_node PRIVATE ${CMAKE_DL_LIBS})
endif()
